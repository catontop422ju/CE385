const scores = [95, 80, ...]; — ประกาศตัวแปรเก็บข้อมูลคะแนนทดสอบทั้ง 13 ค่าในรูปแบบ Array



for (const score of scores) { — เริ่มการวนลูป ดึงตัวเลขจาก Array scores ออกมาทีละค่าแล้วนำไปใส่ไว้ในตัวแปร score

if (score < 0 || score > 100) { — เช็คเงื่อนไขแรก: คะแนนน้อยกว่า 0 หรือ มากกว่า 100 ใช่หรือไม่

console.log("คะแนน " + score + " -> คะแนนไม่ถูกต้อง..."); — แสดงข้อความเตือนเมื่อคะแนนไม่อยู่ในเกณฑ์ 0–100

} else if (score >= 80) { — เช็คต่อ: คะแนนมากกว่าหรือเท่ากับ 80 ใช่หรือไม่ (ช่วง 80–100)

console.log("คะแนน " + score + " -> เกรด A"); — แสดงผลว่าได้เกรด A

} else if (score >= 75) { — เช็คต่อ: คะแนนมากกว่าหรือเท่ากับ 75 ใช่หรือไม่ (ช่วง 75–79)

console.log("คะแนน " + score + " -> เกรด B+"); — แสดงผลว่าได้เกรด B+

} else if (score >= 70) { — เช็คต่อ: คะแนนมากกว่าหรือเท่ากับ 70 ใช่หรือไม่ (ช่วง 70–74)

console.log("คะแนน " + score + " -> เกรด B"); — แสดงผลว่าได้เกรด B

} else if (score >= 65) { — เช็คต่อ: คะแนนมากกว่าหรือเท่ากับ 65 ใช่หรือไม่ (ช่วง 65–69)

console.log("คะแนน " + score + " -> เกรด C+"); — แสดงผลว่าได้เกรด C+

} else if (score >= 60) { — เช็คต่อ: คะแนนมากกว่าหรือเท่ากับ 60 ใช่หรือไม่ (ช่วง 60–64)

console.log("คะแนน " + score + " -> เกรด C"); — แสดงผลว่าได้เกรด C

} else if (score >= 55) { — เช็คต่อ: คะแนนมากกว่าหรือเท่ากับ 55 ใช่หรือไม่ (ช่วง 55–59)

console.log("คะแนน " + score + " -> เกรด D+"); — แสดงผลว่าได้เกรด D+

} else if (score >= 50) { — เช็คต่อ: คะแนนมากกว่าหรือเท่ากับ 50 ใช่หรือไม่ (ช่วง 50–54)

console.log("คะแนน " + score + " -> เกรด D"); — แสดงผลว่าได้เกรด D

} else { — กรณีคะแนนไม่อยู่ในเงื่อนไขด้านบนทั้งหมด (คือช่วง 0–49)

console.log("คะแนน " + score + " -> เกรด F"); — แสดงผลว่าได้เกรด F

} — ปิดขอบเขตการทำงานของ if-else

} — ปิดขอบเขตการทำงานของลูป for...of (เพื่อกลับไปหยิบตัวเลขถัดไปมาทำซ้ำจนครบทุกตัว)
