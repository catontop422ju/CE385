ส่วนประกาศตัวแปร (เก็บข้อมูล)
const name = "Junior"; → สร้างตัวแปรชื่อ name เพื่อเก็บข้อความ "Junior" (ชื่อเล่น)
const studentId = "67111962"; → สร้างตัวแปร studentId เก็บข้อความ "67111962" (รหัสนักศึกษา)
const age = 21; → สร้างตัวแปร age เก็บตัวเลข 21 (อายุ)
const major = "Computer Engineering"; → สร้างตัวแปร major เก็บข้อความชื่อสาขาวิชา
const registerSubject = 6; → สร้างตัวแปร registerSubject เก็บตัวเลข 6 (จำนวนวิชาที่ลง)
const remainYears = 2; → สร้างตัวแปร remainYears เก็บตัวเลข 2 (จำนวนปีที่เหลือจนกว่าจะเรียนจบ)

ส่วนแสดงผลหน้าจอ (พิมพ์ข้อมูลออกมา)
console.log(=====...); → พิมพ์เส้นขอบด้านบนของบัตร
console.log(ชื่อเล่น : ${name}); → พิมพ์คำว่า "ชื่อเล่น :" แล้วดึงค่าจากตัวแปร name (Junior) มาแสดง
console.log(รหัสนักศึกษา : ${studentId}); → พิมพ์คำว่า "รหัสนักศึกษา :" แล้วดึงค่ารหัสมาแสดง *   console.log(อายุ : ${age} ปี); → พิมพ์อายุ พร้อมเติมคำว่า "ปี" ต่อท้าย
console.log(สาขาวิชา : ${major}); → พิมพ์ชื่อสาขาวิชา *   console.log(ลงทะเบียน : ${registerSubject} วิชา); → พิมพ์จำนวนวิชาที่ลงทะเบียน
console.log(ปีที่จะจบ : ${2569 + remainYears}); → พิมพ์ปีที่จะจบ โดยนำเลข 2569 ไปบวกกับตัวแปร remainYears (2) คอมพิวเตอร์จะคำนวณและแสดงผลเป็น 2571 อัตโนมัติ *   console.log(====...); → พิมพ์เส้นขอบปิดท้ายบัตร
