<!-- ส่วนที่ 1: ฟังก์ชัน getMenuPrice (หาราคาตามเมนู) -->

function getMenuPrice(menu) — สร้างฟังก์ชันรับชื่อเมนูเข้ามาตรวจเช็ค

switch (menu) — เริ่มตรวจสอบข้อความชื่อเมนูที่ส่งเข้ามา

case "ข้าวผัด": case "ข้าวมันไก่": case "ข้าวหมูแดง": return 40; — ถ้าเป็น 3 เมนูนี้ ให้คืนค่าราคา 40 บาท (เขียนรวดเดียวเพื่อใช้ราคาร่วมกัน)

case "ผัดไทย": return 50; — ถ้าเป็นผัดไทย ให้คืนค่า 50 บาท

case "ต้มยำกุ้ง": return 60; — ถ้าเป็นต้มยำกุ้ง ให้คืนค่า 60 บาท

default: return 0; — ถ้าเป็นเมนูอื่นๆ ที่ไม่มีในร้าน ให้คืนค่า 0 บาท

<!-- ส่วนที่ 2: ฟังก์ชัน getSizeMultiplier (หาตัวคูณตามขนาด) -->

function getSizeMultiplier(size) — สร้างฟังก์ชันรับชื่อขนาดอาหารเข้ามา

case "ธรรมดา": return 1; — ขนาดธรรมดา คืนค่าตัวคูณ 1 (ราคาปกติ)

case "พิเศษ": return 1.5; — ขนาดพิเศษ คืนค่าตัวคูณ 1.5 (เพิ่มราคา 50%)

case "จัมโบ้": return 2; — ขนาดจัมโบ้ คืนค่าตัวคูณ 2 (เพิ่มราคาเท่าตัว)

default: return 1; — ถ้าใส่ขนาดอื่นนอกเหนือจากนี้ ให้คืนค่าตัวคูณ 1 เป็นค่าเริ่มต้น

const orders = [...] — สร้าง Array เก็บออเดอร์อาหาร 5 รายการ

let totalBill = 0; — ประกาศตัวแปรเก็บราคารวมทั้งบิล เริ่มต้นที่ 0 บาท

for (const order of orders) — วนลูปดึงข้อมูลออเดอร์ออกมาทำทีละรายการ

const price = getMenuPrice(order.menu); — ส่งชื่อเมนูไปหาว่าราคาตั้งต้นกี่บาท

const multiplier = getSizeMultiplier(order.size); — ส่งขนาดไปหาว่าได้ตัวคูณเท่าไหร่

const itemTotal = price * multiplier * order.qty; — คำนวณราคารวมจานนี้ (ราคา × ตัวคูณ × จำนวน)

totalBill += itemTotal; — นำราคาจานนี้ไปบวกสะสมรวมไว้ในบิล
